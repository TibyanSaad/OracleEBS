-- procedure --
CREATE OR REPLACE PROCEDURE TIBYAN_PR_FAC (ERRBUF OUT VARCHAR2,RETCODE OUT NUMBER,NUM IN NUMBER) IS
  FAC NUMBER := 1;
  N NUMBER := NUM;
  e_negative_number  EXCEPTION;
BEGIN

  IF N < 0 THEN
    RAISE e_negative_number;
  END IF;

  WHILE N > 0 LOOP
    FAC := N*FAC;
    N:= N-1;
  END LOOP;

FND_FILE.PUT_LINE(FND_FILE.OUTPUT,' FACTORIAL IS ' || FAC);

EXCEPTION
  WHEN NO_DATA_FOUND THEN
    FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'ERROR: Please enter a number.');
  WHEN e_negative_number THEN
    FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'Error: factorial is not defined for negative numbers.');
  WHEN OTHERS THEN
    FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'ERROR: CHECK YOUR PROGRAM');
END;
/
