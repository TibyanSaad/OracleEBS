-- EBS PROCEDURE--
CREATE OR REPLACE PROCEDURE TIBYAN_PR_EBS (ERRBUF OUT VARCHAR2,
  RETCODE OUT NUMBER) IS
BEGIN
  FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'WELCOME TO ORACLE EBS PR...!!!');
END;
/

-- STEPS--
Create an Oracle EBS procedure & register it in EBS
1. login to EBS
2. Create an executable: navigate to APPLICATION DEVELOPER> CONCURRENT> EXECUTABLE
            Name: TIBYAN_PR_EBS
            Short Name: TIBYAN_PR_EBS
            Application: AOL (Application Object Library) <tab>
            Execution Method: PL/SQL Stored Procedure
            Execution File Name: TIBYAN_PR_EBS
                            <SAVE>
3. TO CHECK IF IT IS CREATED GO TO View > Query by example > Enter
            Name: TIBYAN_PR_EBS
4. THEN View > Query by example > Run AND YOU CAN SEE THE PROCEDURE DETAILS
5. create a program: navigate to APPLICATION DEVELOPER> CONCURRENT> PROGRAM
            Name: TIBYAN_PR_EBS
            Short Name: TIBYAN_PR_EBS
            Application: AOL (application object library) <Tab>
            Executable: TIBYAN_PR_EBS
            Output: HTML
                            <SAVE>
6. Create a request group: navigate to System Administrator> Security>Responsibility>Request
            Name: TIBYAN_RG_PGMS
            Application: AOL (application object library) <Tab>
            Program: TIBYAN_PR_EBS
                            <SAVE>
7.CREATE A NEW USER RESPONSIBILITY: NAVIGATE TO  System Administrator> Security> Responsibility> Define
            Name: TIBYAN_RESP
            Application: AOL (application object library) <Tab>
            Key: TIBYAN_RESP_KEY
            Name: Standard
            Menu: Position Control, Vision University
                            <SAVE>
8. Assign the responsibility to a user: System Administrator> Security> User> Define
            view> query by example> enter
                    user: TIBYAN
            view> query by example> run
       add the responsibility
            TIBYAN_RESP
9. create a request: switch to your responsibility then navigate to View > Request >Submit a New Request >Single (Default) >OK
            CLICK THE LOV To select a Program
            TIBYAN_PR_EBS then Submit
            Find> View Output
