-- EBS PROCEDURE --
CREATE OR REPLACE PROCEDURE TIBYAN_PR_SYSDT(ERRBUF OUT VARCHAR2,
    RETCODE OUT NUMBER) IS
V_DT DATE;
BEGIN
  SELECT SYSDATE INTO V_DT FROM DUAL;
  FND_FILE.PUT_LINE(FND_FILE.LOG,'YOU CAN SEE LOG DETAILS HERE!');
  FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'CURRENT DATE =' || V_DT);
EXCEPTION
  WHEN OTHERS THEN
  FND_FILE.PUT_LINE(FND_FILE.OUTPUT,'CHECK YOUR PROGRAM: '
                      || ERRBUF||'ERRCODE='||RETCODE);
END;
/

-- STEPS --
1.login into EBS
2. Create an executable: navigate to APPLICATION DEVELOPER> CONCURRENT> EXECUTABLE
            Name: TIBYAN_PR_SYSDT
            Short Name: TIBYAN_PR_SYSDT
            Application: AOL (Application Object Library) <tab>
            Execution Method: PL/SQL Stored Procedure
            Execution File Name: TIBYAN_PR_SYSDT
                            <SAVE>
3. TO CHECK IF IT IS CREATED GO TO View > Query by example > Enter
            Name: TIBYAN_PR_SYSDT
4. THEN View > Query by example > Run AND YOU CAN SEE THE PROCEDURE DETAILS
5. create a program: navigate to APPLICATION DEVELOPER> CONCURRENT> PROGRAM
            Name: TIBYAN_PR_SYSDT_PG
            Short Name: TIBYAN_PR_SYSDT_PG
            Application: AOL (application object library) <Tab>
            Executable: TIBYAN_PR_SYSDT_PG
            Output: HTML
                            <SAVE>
6. Add it to the already existing request group: navigate to System Administrator> Security>Responsibility>Request
            Name: TIBYAN_RG_PGMS
            Application: AOL (application object library) <Tab>
            Program: TIBYAN_PR_SYSDT_PG
                            <SAVE>
7. create a request: switch to your responsibility then navigate to View > Request >Submit a New Request >Single (Default) >OK
            CLICK THE LOV To select a Program
            TIBYAN_PR_EBS then Submit
            Find> View Output
